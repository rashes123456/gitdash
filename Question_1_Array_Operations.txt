QUESTION 1 - ARRAY OPERATIONS USING A MENU
==========================================

/*
   This program performs common operations on an integer array.
   The user can insert, delete, search, reverse and calculate
   different values using the menu.
*/

#include <stdio.h>

#define MAX 100

void insertElement(int arr[], int *n, int allowedMax)
{
    int index, value, i;

    if (*n >= MAX)
    {
        printf("There is no space left in the array.\n");
        return;
    }

    printf("Enter the index for insertion: ");
    scanf("%d", &index);

    if (index < 0 || index > *n)
    {
        printf("That index is not valid.\n");
        return;
    }

    printf("Enter the value (1 to %d): ", allowedMax);
    scanf("%d", &value);

    if (value < 1 || value > allowedMax)
    {
        printf("Value must be between 1 and %d.\n", allowedMax);
        return;
    }

    /* Shift the elements right to make an empty position. */
    for (i = *n; i > index; i--)
    {
        arr[i] = arr[i - 1];
    }

    arr[index] = value;
    (*n)++;

    printf("Element inserted.\n");
}

void deleteElement(int arr[], int *n)
{
    int index, i;

    if (*n == 0)
    {
        printf("Array is empty.\n");
        return;
    }

    printf("Enter the index to delete: ");
    scanf("%d", &index);

    if (index < 0 || index >= *n)
    {
        printf("Invalid index.\n");
        return;
    }

    /* Fill the deleted position by shifting later elements left. */
    for (i = index; i < *n - 1; i++)
    {
        arr[i] = arr[i + 1];
    }

    (*n)--;
    printf("Element deleted.\n");
}

void showMinimum(int arr[], int n)
{
    int i, minimum;

    if (n == 0)
    {
        printf("Array is empty.\n");
        return;
    }

    minimum = arr[0];

    for (i = 1; i < n; i++)
    {
        if (arr[i] < minimum)
            minimum = arr[i];
    }

    printf("Minimum value = %d\n", minimum);
}

void showMaximum(int arr[], int n)
{
    int i, maximum;

    if (n == 0)
    {
        printf("Array is empty.\n");
        return;
    }

    maximum = arr[0];

    for (i = 1; i < n; i++)
    {
        if (arr[i] > maximum)
            maximum = arr[i];
    }

    printf("Maximum value = %d\n", maximum);
}

void displayArray(int arr[], int n)
{
    int i;

    if (n == 0)
    {
        printf("Array is empty.\n");
        return;
    }

    printf("Array: ");
    for (i = 0; i < n; i++)
        printf("%d ", arr[i]);

    printf("\n");
}

void displayReverse(int arr[], int n)
{
    int i;

    printf("Array from last to first: ");
    for (i = n - 1; i >= 0; i--)
        printf("%d ", arr[i]);

    printf("\n");
}

void searchValue(int arr[], int n)
{
    int value, i, found = 0;

    printf("Enter the value to search: ");
    scanf("%d", &value);

    for (i = 0; i < n; i++)
    {
        if (arr[i] == value)
        {
            printf("Found at index %d\n", i);
            found = 1;
        }
    }

    if (found == 0)
        printf("Value was not found.\n");
}

void countValue(int arr[], int n)
{
    int value, count = 0, i;

    printf("Enter the value whose count you want: ");
    scanf("%d", &value);

    for (i = 0; i < n; i++)
    {
        if (arr[i] == value)
            count++;
    }

    printf("%d appears %d time(s).\n", value, count);
}

void showAverage(int arr[], int n)
{
    int i, sum = 0;
    float average;

    if (n == 0)
    {
        printf("Array is empty.\n");
        return;
    }

    for (i = 0; i < n; i++)
        sum = sum + arr[i];

    average = (float)sum / n;
    printf("Average = %.2f\n", average);
}

void findDuplicates(int arr[], int n)
{
    int i, j, k, alreadyPrinted, found = 0;

    for (i = 0; i < n; i++)
    {
        alreadyPrinted = 0;

        /* Check whether this value appeared before. */
        for (k = 0; k < i; k++)
        {
            if (arr[k] == arr[i])
            {
                alreadyPrinted = 1;
                break;
            }
        }

        if (alreadyPrinted == 1)
            continue;

        for (j = i + 1; j < n; j++)
        {
            if (arr[i] == arr[j])
            {
                printf("Duplicate value: %d\n", arr[i]);
                found = 1;
                break;
            }
        }
    }

    if (found == 0)
        printf("No duplicate values found.\n");
}

void reverseArray(int arr[], int n)
{
    int left = 0, right = n - 1, temp;

    while (left < right)
    {
        temp = arr[left];
        arr[left] = arr[right];
        arr[right] = temp;

        left++;
        right--;
    }

    printf("Array has been reversed.\n");
}

int main()
{
    int arr[MAX];
    int n, originalN, i, choice;

    printf("Enter the number of elements: ");
    scanf("%d", &n);

    if (n <= 0 || n > MAX)
    {
        printf("Please enter a size between 1 and %d.\n", MAX);
        return 1;
    }

    originalN = n;

    printf("Enter %d values (from 1 to %d):\n", n, originalN);
    for (i = 0; i < n; i++)
    {
        scanf("%d", &arr[i]);

        if (arr[i] < 1 || arr[i] > originalN)
        {
            printf("Invalid value. Values must be from 1 to %d.\n", originalN);
            return 1;
        }
    }

    do
    {
        printf("\n========== ARRAY MENU ==========\n");
        printf("1. Insert an element\n");
        printf("2. Delete an element\n");
        printf("3. Find minimum\n");
        printf("4. Find maximum\n");
        printf("5. Display array\n");
        printf("6. Display in reverse order\n");
        printf("7. Search for a value\n");
        printf("8. Count a value\n");
        printf("9. Find average\n");
        printf("10. Find duplicate values\n");
        printf("11. Reverse the actual array\n");
        printf("12. Exit\n");
        printf("Enter your choice: ");
        scanf("%d", &choice);

        switch (choice)
        {
            case 1:
                insertElement(arr, &n, originalN);
                break;

            case 2:
                deleteElement(arr, &n);
                break;

            case 3:
                showMinimum(arr, n);
                break;

            case 4:
                showMaximum(arr, n);
                break;

            case 5:
                displayArray(arr, n);
                break;

            case 6:
                displayReverse(arr, n);
                break;

            case 7:
                searchValue(arr, n);
                break;

            case 8:
                countValue(arr, n);
                break;

            case 9:
                showAverage(arr, n);
                break;

            case 10:
                findDuplicates(arr, n);
                break;

            case 11:
                reverseArray(arr, n);
                break;

            case 12:
                printf("Closing the program.\n");
                break;

            default:
                printf("Please choose a valid option.\n");
        }

    } while (choice != 12);

    return 0;
}
